class Seat:
    def __init__(self, number, price):
        self.number = number
        self.available = True
        self.gender = None
        if self.number <= 14:
            self.price = price
        else:
            self.price = price // 2 + 20

    def show(self):
        if self.available:
            status = "available"
        elif self.gender == "female_only":
            status = "female only"
        else:
            status = "booked " + str(self.gender)
        print("Seat", self.number, "|", status, "|", "Price", self.price)

    seats = []

def find(seats, n):
    for s in seats:
        if s.number == n:
            return s
    return None

# Seat choosing requirement
def seat_booking(seats, From, To):
    show_seats(seats, From, To)
    try:
        seat_no = int(input("Enter Seat No: "))
    except ValueError:
        print("Invalid Seat")
        return None

    chosen = find(seats, seat_no)
    if chosen is None:
        print("Invalid Seat")
        return None
    if not chosen.available and chosen.gender != "female_only":
        print("Seat", seat_no, "is Already Booked")
        return None
    return seat_no, chosen

    if chosen.gender == "female_only" and gender != "female":

        return False

    ticket = ticket_create(seat_no)
    if ticket is None:
        print("Seat not confirmed because payment failed")
        return False
    
    chosen.available = False
    chosen.gender = gender
    if gender == "female":
        if seat_no % 2 == 1:
            pair = seat_no + 1
        else:
            pair = seat_no - 1
        next_seat = find(seats, pair)
        if next_seat is not None and next_seat.available:
            next_seat.available = False
            next_seat.gender = "female_only"
            print("Seat", next_seat.number, "reserved female only")
    print("Confirmed")
    return True

# BOOK TICKET
def book_ticket(bus, route_id):
    seats = load_seats(bus.bus_id)
    show_seats(seats)

    try:
        seat_no = int(input("\nEnter seat number: ").strip())
    except ValueError:
        print("Invalid seat number!")
        return

    chosen = find_seat(seats, seat_no)

    if chosen is None:
        print("Seat not found!")
        return

    if chosen.available is False and chosen.status != "female_only":
        print("Seat already booked!")
        return

    passenger = get_passenger_details()

    if chosen.status == "female_only" and passenger.gender != "female":
        print("This seat is for female passengers only!")
        return

    payment_type = do_payment()
    if payment_type is None:
        print("Payment not completed. Booking cancelled.")
        return

    ticket = Ticket(seat_no, bus.bus_id)

    try:
        update_seat(chosen, passenger.gender, seats)
        save_booking(passenger, ticket, bus.bus_id, route_id, payment_type)
        db.commit()
    except mysql.connector.Error as err:
        db.rollback()
        print("Booking failed:", err)
        return

    print(" Ticket Booked Successfully! ")
    print("Name: ", passenger.name)
    print("Age: ", passenger.age)
    print("Gender: ", passenger.gender)
    print("Mobile: ", passenger.mobileno)
    print("Seat No: ", ticket.seat_no)
    ticket.show()

    print(" Updated Seat List ")
    updated = load_seats(bus.bus_id)
    show_seats(updated)

# UPDATE SEAT AFTER BOOKING
def update_seat(chosen, gender, seats):
    chosen.available = False
    chosen.status  = "booked " + gender

    cursor.execute(
        "update seat set is_available=%s, seat_status=%s where seat_id=%s",
        (chosen.available, chosen.status, chosen.seat_id)
    )

    if gender == "female":
        if chosen.number % 2 == 1:
            pair = chosen.number + 1
        else:
            pair = chosen.number - 1

        next_seat = find_seat(seats, pair)

        if next_seat is not None and next_seat.available:
            cursor.execute(
                "update seat set is_available=false, seat_status='female_only' where seat_id=%s",
                (next_seat.seat_id,)
            )
            print("Seat", pair, "is now reserved for female passengers.")

def save_booking(passenger, ticket, bus_id, route_id, payment_type):
    cursor.execute(
        "insert into passengers(name_, age, gender, route_id, mobile, email) values(%s, %s, %s, %s, %s, %s)",

    )

    passenger_id = cursor.lastrowid
    payment_value = payment_type + "-" + str(passenger_id)

    cursor.execute(
        "insert into booking(passenger_id, bus_id, seat_no, name_, pnr, ticket_no, payment_type) values(%s, %s, %s, %s, %s, %s, %s)",
        (passenger_id, bus_id, ticket.seat_no, passenger.name, ticket.pnr, ticket.ticket_no, payment_value)
    )

# VERIFY TICKET
def verify_ticket():
    print(" Ticket Verification ")

    name = input("Your Name: ").strip()
    ticket_no = input("Your Ticket No: ").strip()
    seat_no = input("Your Seat No: ").strip()
    pnr = input("Your PNR: ").strip()

    cursor.execute(
        """
        select p.name_, b.ticket_no, b.seat_no, b.pnr, r.to_, r.arrival
        from booking b
        join passengers p on b.passenger_id = p.passenger_id
        join bus bu on b.bus_id = bu.bus_id
        join routes r on bu.route_id = r.route_id
        where p.name_=%s and b.ticket_no=%s and b.seat_no=%s and b.pnr=%s
        """,
        (name, ticket_no, seat_no, pnr)
    )
    row = cursor.fetchone()

    if row is None:
        print("Ticket not found! Please check your details.")
        return

    print("\nVerification Successful!")
    print("You can board the bus.")
    print("Destination : ", row[4])
    print("Arrival Time : ", row[5])
    print("Have a safe journey!")

def cancel_ticket():

    name = input("Your Name: ")
    ticket_no = input("Your Ticket No: ")

    pnr = input("Your PNR: ")

    cursor.execute(
        """
        select b.booking_id, b.passenger_id, b.bus_id, b.seat_no, p.gender
        from booking b
        join passengers p on b.passenger_id = p.passenger_id
        where p.name_=%s and b.ticket_no=%s and b.seat_no=%s and b.pnr=%s
        """,
        (name, ticket_no, seat_no, pnr)
    )
    row = cursor.fetchone()

    if row is None:
        print("Ticket not found!")
        return

    booking_id   = row[0]
    passenger_id = row[1]
    bus_id = row[2]
    booked_seat = row[3]
    gender = row[4]

    confirm = input("Are you sure you want to cancel? (yes/no): ").lower()
    if confirm != "yes":
        print("Cancellation stopped.")
        return
    try:
        cursor.execute("delete from booking where booking_id=%s", (booking_id,))
        cursor.execute("delete from passengers where passenger_id=%s", (passenger_id,))
        cursor.execute(
            "update seat set is_available=true, seat_status='available' where bus_id=%s and seat_no=%s",
            (bus_id, booked_seat)
        )

        if gender == "female":
            if booked_seat % 2 == 1:
                pair = booked_seat + 1
            else:
                pair = booked_seat - 1

            cursor.execute(
                "update seat set is_available=true, seat_status='available' where bus_id=%s and seat_no=%s and seat_status='female_only'",
                (bus_id, pair)
            )

        db.commit()
        print("Ticket cancelled successfully!")

    except mysql.connector.Error as err:
        db.rollback()
        print("Cancellation failed:", err)
